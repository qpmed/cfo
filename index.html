# ═══════════════════════════════════════════════════
#  MT5 → WebSocket Bridge  |  Forex Signal Bot
#  Yêu cầu: pip install MetaTrader5 websockets
# ═══════════════════════════════════════════════════
import asyncio, json, datetime
import MetaTrader5 as mt5
import websockets

# ── CẤU HÌNH ────────────────────────────────────────
WS_HOST = "localhost"
WS_PORT = 8765
MT5_LOGIN  = 0          # ← Điền MT5 login (XM demo)
MT5_PASSWORD = ""       # ← Điền mật khẩu
MT5_SERVER   = ""       # ← Điền tên server (vd: XMGlobal-MT5 2)
SYMBOLS = [
    "EURUSD","GBPUSD","USDJPY","AUDUSD","USDCAD",
    "EURGBP","NZDUSD","USDCHF","EURJPY","GBPJPY"
]
INTERVAL_SEC = 1        # Gửi giá mỗi 1 giây
# ────────────────────────────────────────────────────

clients = set()
price_cache = {}

def connect_mt5():
    if not mt5.initialize(login=MT5_LOGIN,
                          password=MT5_PASSWORD,
                          server=MT5_SERVER):
        raise RuntimeError(f"MT5 init failed: {mt5.last_error()}")
    print(f"[MT5] Connected: {mt5.terminal_info().name}")

def get_prices():
    data = {}
    for sym in SYMBOLS:
        tick = mt5.symbol_info_tick(sym)
        if tick:
            info = mt5.symbol_info(sym)
            digits = info.digits if info else 5
            data[sym] = {
                "bid": round(tick.bid, digits),
                "ask": round(tick.ask, digits),
                "time": tick.time,
                "digits": digits
            }
    return data

def get_indicators(sym):
    """Tính RSI(14), MACD, EMA trend từ dữ liệu M15."""
    rates = mt5.copy_rates_from_pos(sym, mt5.TIMEFRAME_M15, 0, 100)
    if rates is None or len(rates) < 30:
        return None
    closes = [r['close'] for r in rates]
    # RSI 14
    gains, losses = [], []
    for i in range(1, 15):
        d = closes[-i] - closes[-i-1]
        (gains if d > 0 else losses).append(abs(d))
    avg_g = sum(gains)/14 if gains else 0.0001
    avg_l = sum(losses)/14 if losses else 0.0001
    rsi = round(100 - 100/(1 + avg_g/avg_l), 2)
    # EMA 20 vs EMA 50
    def ema(prices, n):
        k = 2/(n+1); e = prices[0]
        for p in prices[1:]: e = p*k + e*(1-k)
        return e
    ema20 = ema(closes[-50:], 20)
    ema50 = ema(closes[-50:], 50)
    ema_trend = "Bullish" if ema20 > ema50 else "Bearish"
    # MACD (12,26,9)
    macd_val = round(ema(closes[-50:], 12) - ema(closes[-50:], 26), 5)
    return {"rsi": rsi, "ema": ema_trend, "macd": macd_val}

async def broadcast_prices():
    while True:
        try:
            prices = get_prices()
            price_cache.update(prices)
            payload = {
                "type": "prices",
                "data": prices,
                "ts": datetime.datetime.utcnow().isoformat()
            }
            msg = json.dumps(payload)
            if clients:
                await asyncio.gather(
                    *[c.send(msg) for c in list(clients)],
                    return_exceptions=True
                )
        except Exception as e:
            print(f"[BROADCAST] {e}")
        await asyncio.sleep(INTERVAL_SEC)

async def handler(ws):
    clients.add(ws)
    print(f"[WS] Client connected ({len(clients)} total)")
    # Send welcome + current prices
    await ws.send(json.dumps({"type":"connected","version":"1.0"}))
    if price_cache:
        await ws.send(json.dumps({"type":"prices","data":price_cache}))
    try:
        async for raw in ws:
            try:
                msg = json.loads(raw)
                # HTML app gửi yêu cầu indicators
                if msg.get("type") == "scan":
                    result = {}
                    for sym in msg.get("symbols", SYMBOLS):
                        ind = get_indicators(sym)
                        if ind:
                            tick = mt5.symbol_info_tick(sym)
                            info = mt5.symbol_info(sym)
                            digits = info.digits if info else 5
                            result[sym] = {
                                **ind,
                                "bid": round(tick.bid, digits) if tick else 0,
                                "ask": round(tick.ask, digits) if tick else 0
                            }
                    await ws.send(json.dumps({"type":"scan_result","data":result}))
            except Exception as e:
                print(f"[MSG] {e}")
    except websockets.exceptions.ConnectionClosed:
        pass
    finally:
        clients.discard(ws)
        print(f"[WS] Client disconnected ({len(clients)} total)")

async def main():
    connect_mt5()
    print(f"[WS] Server starting on ws://{WS_HOST}:{WS_PORT}")
    async with websockets.serve(handler, WS_HOST, WS_PORT):
        await asyncio.gather(broadcast_prices(), asyncio.Future())

if __name__ == "__main__":
    asyncio.run(main())
